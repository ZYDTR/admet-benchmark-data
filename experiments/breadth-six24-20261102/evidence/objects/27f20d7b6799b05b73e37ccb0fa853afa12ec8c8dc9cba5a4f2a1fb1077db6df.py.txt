class Storage:

class Tensor:
 def __init__(self,storage,offset,shape,stride,*args):self.storage,self.offset,self.shape,self.stride=storage,offset,tuple(shape),tuple(stride)

 def values(self):
  fmt={'FloatStorage':'f','DoubleStorage':'d','LongStorage':'q','IntStorage':'i','ByteStorage':'B','BoolStorage':'?'}[self.storage.dtype]
  return [x[0] for x in struct.iter_unpack('<'+fmt,self.raw)]
class Reader(pickle.Unpickler):
 def find_class(self,module,name):
  if (module,name)==('collections','OrderedDict'):return collections.OrderedDict
  if (module,name)==('torch._utils','_rebuild_tensor_v2'):return Tensor
  if module=='torch' and name in ('FloatStorage','DoubleStorage','LongStorage','IntStorage','ByteStorage','BoolStorage'):return name
  raise ValueError('Rejected global '+module+'.'+name)

  visit(result);return result
