def digest_model(model):
 import torch
 kinds={n:'parameter' for n,_ in model.named_parameters()};kinds.update({n:'buffer' for n,_ in model.named_buffers()})
 result={}
 for n,t in model.state_dict().items():
  data=t.detach().cpu().contiguous().reshape(-1).view(torch.uint8).numpy().tobytes()
  result[n]={'sha256':hashlib.sha256(data).hexdigest(),'shape':list(t.shape),'dtype':str(t.dtype),'kind':kinds[n]}
 return result

def digest_state(state):
 import torch
 return {k:hashlib.sha256(v.detach().cpu().contiguous().reshape(-1).view(torch.uint8).numpy().tobytes()).hexdigest() for k,v in state.items()}

def equal(a,b):
 import torch
 if isinstance(a,torch.Tensor):return isinstance(b,torch.Tensor) and a.dtype==b.dtype and a.shape==b.shape and torch.equal(a.cpu(),b.cpu())
 if type(a)!=type(b):return False
 if isinstance(a,dict):return a.keys()==b.keys() and all(equal(a[k],b[k]) for k in a)
 if isinstance(a,(list,tuple)):return len(a)==len(b) and all(equal(x,y) for x,y in zip(a,b))
 return a==b

def restore_rng(value,gen):
 import torch,numpy as np
 random.setstate(value['python']);n=value['numpy'];np.random.set_state((n['name'],n['keys'].numpy().astype(np.uint32),n['pos'],n['has_gauss'],n['cached_gaussian']))
 torch.set_rng_state(value['torch_cpu']);torch.cuda.set_rng_state_all(value['torch_cuda']);gen.set_state(value['sampler'])
