from pathlib import Path
import json, sys, time, hainan
P = Path(__file__).resolve().parent
task = json.loads((P.parent / 'TASK.json').read_text())
index = int(sys.argv[1])
attempt = int(sys.argv[2])
run = task['runs'][index]['run_id']
ledger = json.loads((P / 'compute-ledger.json').read_text())
assert not any((x['run_id'] == run and x.get('complete') for x in ledger['attempts']))
remaining = ledger['allowance'] - sum((x['elapsed_seconds'] for x in ledger['attempts']))
assert remaining > 30
source = (P / 'run-one-remote.py').read_text().replace('RUN_NAME', repr(run)).replace('TIME_LIMIT', str(min(900, remaining))).replace('root=BASE/run;', 'root=BASE/run' + ("/'attempt-%02d'" % attempt if attempt > 1 else '') + ';').replace('source/compute_paced.py', 'source/compute_cached.py')
source = source.replace('support/thread-names.so', 'support/thread-names-v2.so')
tag = f'fit-{index}-{attempt}'
(P / f'{tag}-remote.py').write_text(source)
try:
    raw = hainan.once(source, tag, timeout=min(900, remaining) + 45)
except Exception:
    if (P / 'raw' / f'{tag}.stdout').exists():
        try:
            raw = (P / 'raw' / f'{tag}.stdout').read_bytes()
            v = json.loads(raw)
            assert 'actual_strace_wait_exit' in v
        except Exception:
            raise
    else:
        raise
else:
    v = json.loads(raw)
v.update(attempt=attempt, output_relative=run + ('/attempt-%02d' % attempt if attempt > 1 else ''))
ledger['attempts'].append(v)
(P / 'compute-ledger.json').write_text(json.dumps(ledger, indent=2) + '\n')
(P / f'{tag}-result.json').write_text(json.dumps(v, indent=2) + '\n')
print(json.dumps(v))
