lib = ctypes.CDLL(None, use_errno=True)
lib.syscall.restype = ctypes.c_long
abi = lib.syscall(444, ctypes.c_void_p(), ctypes.c_size_t(0), ctypes.c_uint(1))
assert abi >= 1, (abi, ctypes.get_errno())
handled = sum((1 << i for i in [1, *range(4, 13)]))
if abi >= 2:
    handled |= 1 << 13
if abi >= 3:
    handled |= 1 << 14

class Ruleset(ctypes.Structure):
    _fields_ = [('handled_access_fs', ctypes.c_uint64)]

class Beneath(ctypes.Structure):
    _pack_ = 1
    _fields_ = [('allowed_access', ctypes.c_uint64), ('parent_fd', ctypes.c_int32)]
attr = Ruleset(handled)
fd = lib.syscall(444, ctypes.byref(attr), ctypes.sizeof(attr), 0)
assert fd >= 0, ctypes.get_errno()

def allow(path, access):
    descriptor = os.open(path, os.O_PATH | os.O_CLOEXEC)
    try:
        rule = Beneath(access, descriptor)
        assert lib.syscall(445, fd, 1, ctypes.byref(rule), 0) == 0, (str(path), ctypes.get_errno())
    finally:
        os.close(descriptor)
allow(ROOT, handled)
devices = []
for pattern in ('nvidia[0-9]*', 'nvidiactl', 'nvidia-uvm', 'nvidia-uvm-tools', 'nvidia-modeset'):
    for path in Path('/dev').glob(pattern):
        if path.is_char_device():
            allow(path, 1 << 1)
            devices.append(str(path))
allow(Path('/dev/null'), 1 << 1)
assert lib.prctl(38, 1, 0, 0, 0) == 0
assert lib.syscall(446, fd, 0) == 0, ctypes.get_errno()
os.close(fd)

import csv, importlib.metadata
CACHE = Path('<AUTHORIZED_PROJECT_ROOT>/admet/agent-led-resume-20261102/hainan-agentled-mse-s44-20261102/attempt-03')
cache_hashes = json.loads((ROOT / 'source/cache-reference.json').read_text())

class CachedEncoder(science.CheMeleonEmbedder):

    def embed(self, smiles):
        self.assert_frozen()
        for split in ('train', 'valid', 'test'):
            with (ROOT / 'data' / ('train.csv' if split == 'train' else split + '.csv')).open(newline='', encoding='utf-8-sig') as f:
                rows = list(csv.DictReader(f))
            if smiles != [r['Drug'] for r in rows]:
                continue
            data_path = CACHE / 'artifacts' / ('features-' + split + '.npz')
            id_path = CACHE / 'artifacts' / ('features-' + split + '-identity.json')
            for path in (data_path, id_path):
                assert path.resolve(strict=True) == path and path.stat().st_nlink == 1
                assert __import__('hashlib').sha256(path.read_bytes()).hexdigest() == cache_hashes[str(path.relative_to(CACHE))]
            identity = json.loads(id_path.read_text())
            meta = identity['encoder']
            assert meta['encoder_py_sha256'] == __import__('hashlib').sha256((ROOT / 'encoder.py').read_bytes()).hexdigest()
            assert meta['checkpoint_sha256'] == self.asserted_checkpoint_sha256 and meta['feature_dim'] == self.feature_dim
            for name, version in meta['dependencies'].items():
                assert importlib.metadata.version(name) == version
            with science.np.load(data_path, allow_pickle=False) as f:
                features = f['features'].copy()
                mask = f['parse_mask'].copy()
            actual = science.feature_identity(features, mask, rows, meta, self.feature_dim, __import__('hashlib').sha256((ROOT / 'data' / (split + '.csv')).read_bytes()).hexdigest())
            assert actual == identity, 'cached feature content/row/encoder mismatch'
            count = len(list((ROOT / 'raw').glob('feature-cache-use-*.json')))
            write_json(ROOT / 'raw' / ('feature-cache-use-%02d.json' % count), {'source_root': str(CACHE), 'split': split, 'cache_identity': identity['identity_sha256'], 'features_sha256': identity['features']['sha256'], 'no_new_encoder_forward': True, 'verified_content_and_full_row_identity': True})
            return (features, mask)
        raise ValueError('no exact cache row sequence match')
science.CheMeleonEmbedder = CachedEncoder
sys.modules['encoder'].CheMeleonEmbedder = CachedEncoder
original_step = torch.optim.AdamW.step

def paced_step(self, *args, **kwargs):
    value = original_step(self, *args, **kwargs)
    torch.cuda.synchronize()
    time.sleep(0.2)
    return value
torch.optim.AdamW.step = paced_step
