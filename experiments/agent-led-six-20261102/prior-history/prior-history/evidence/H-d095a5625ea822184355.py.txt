def diff(a,b,path=''):
 if isinstance(a,dict) and isinstance(b,dict):
  out=[]
  for k in set(a)|set(b):
   if k not in a or k not in b:out.append(path+'.'+k)
   else:out+=diff(a[k],b[k],path+'.'+k)
  return out
 return [] if a==b else [path]
