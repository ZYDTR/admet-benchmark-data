"""Manual ledger of web-rendered text actually displayed/read; not network retrieval."""
import json
from pathlib import Path
P=Path(__file__).with_name('reading-coverage.json')
def load():return json.loads(P.read_text()) if P.exists() else {}
def add(key,total,start,end,ref):
    d=load();r=d.setdefault(key,{'total_web_lines':total,'segments':[]});assert r['total_web_lines']==total
    z={'start':start,'end':end,'web_ref':ref};assert 0<=start<=end<total
    if z not in r['segments']:r['segments'].append(z)
    P.write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n')
def gaps(key):
    d=load()[key];seen=set()
    for s in d['segments']:seen.update(range(s['start'],s['end']+1))
    missing=sorted(set(range(d['total_web_lines']))-seen);out=[]
    for i in missing:
        if not out or i>out[-1][1]+1:out.append([i,i])
        else:out[-1][1]=i
    return out
if __name__=='__main__':
    for k,v in load().items():print(k,v['total_web_lines'],'missing',gaps(k))
