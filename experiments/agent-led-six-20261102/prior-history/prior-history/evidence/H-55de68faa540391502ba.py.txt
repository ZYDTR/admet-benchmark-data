def canonical(root,path,mountinfo=None):
    assert root.resolve(strict=True)==root and (path==root or root in path.parents)
    assert path.resolve(strict=False)==path
    for parent in (path,*path.parents):
        if parent.exists() or parent.is_symlink():
            info=parent.lstat();assert not stat.S_ISLNK(info.st_mode)
            if stat.S_ISREG(info.st_mode):assert info.st_nlink==1
    if mountinfo is None:
        file=Path('/proc/self/mountinfo');mountinfo=file.read_text() if file.exists() else ''
    for line in mountinfo.splitlines():
        fields=line.split();assert len(fields)>=6
        mount=Path(re.sub(r'\\([0-7]{3})',lambda m:chr(int(m[1],8)),fields[4]))
        assert not ((mount==root or root in mount.parents) and (mount==path or mount in path.parents))
    return path

def collect(blob,manifest,digest):
    assert 0<len(manifest)<=7 and len(blob)<=4*1024**2 and hashlib.sha256(blob).hexdigest()==digest
    for name,item in manifest.items():
        assert name.startswith(PREFIX) and name[len(PREFIX):] in ALLOWED and 0<=item['size']<=1024**2
    with zipfile.ZipFile(io.BytesIO(blob)) as archive:
        assert len(archive.namelist())==len(manifest) and set(archive.namelist())==set(manifest)
        result={}
        for item in archive.infolist():
            mode=stat.S_IFMT(item.external_attr>>16)
            assert not item.is_dir() and mode in (0,stat.S_IFREG) and not item.flag_bits&1
            assert item.file_size==manifest[item.filename]['size']
            value=archive.read(item.filename);assert len(value)==manifest[item.filename]['size'] and hashlib.sha256(value).hexdigest()==manifest[item.filename]['sha256'];result[item.filename]=value
    return result

def read_frame(stream,size):
    assert 0<size<=4*1024**2
    value=stream.read(size+1);assert len(value)==size,'truncated or extra transport bytes'
    return value

def append(base,data):
    for name in data:
        assert name.startswith(PREFIX) and name[len(PREFIX):] in ALLOWED
        path=canonical(base,base/name);assert not path.exists() and not path.is_symlink()
    for name,value in data.items():
        path=canonical(base,base/name);path.parent.mkdir(parents=True,exist_ok=True);canonical(base,path.parent)
        with path.open('xb') as stream:stream.write(value);stream.flush();os.fsync(stream.fileno())
