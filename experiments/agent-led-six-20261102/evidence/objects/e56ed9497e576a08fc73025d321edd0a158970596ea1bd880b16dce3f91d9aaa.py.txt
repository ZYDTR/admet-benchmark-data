def stamp():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def ident(pid):
    raw = (Path('/proc') / str(pid) / 'stat').read_text()
    v = raw.rsplit(')', 1)[1].split()
    return {'pid': pid, 'start_ticks': v[19], 'pgid': int(v[2]), 'state': v[0], 'cpu_ticks': int(v[11]) + int(v[12]), 'start_stat': raw}

def dump(name, value):
    with (root / 'raw' / name).open('x') as f:
        json.dump(value, f, indent=2)
        f.flush()
        os.fsync(f.fileno())

def query(argv):
    at = stamp()
    try:
        p = subprocess.run(argv, capture_output=True, text=True, timeout=8)
        v = {'argv': argv, 'at': at, 'actual_exit': p.returncode, 'stdout': p.stdout, 'stderr': p.stderr}
    except Exception as e:
        v = {'argv': argv, 'at': at, 'actual_exit': None, 'error': repr(e)}
    with (root / 'raw/resource-observations.jsonl').open('a') as f:
        f.write(json.dumps(v) + '\n')
        f.flush()
        os.fsync(f.fileno())
    if v['actual_exit'] != 0:
        raise RuntimeError('resource query unknown')
    return v['stdout']

def health():
    g = ET.fromstring(query(['/usr/bin/nvidia-smi', '-i', uuid, '-q', '-x'])).find('gpu')
    for key, value in {'uuid': uuid, 'gpu_recovery_action': 'None', 'compute_mode': 'Default', 'remapped_rows/remapped_row_pending': 'No', 'remapped_rows/remapped_row_failure': 'No', 'ecc_errors/volatile/sram_uncorrectable_parity': '0', 'ecc_errors/volatile/sram_uncorrectable_secded': '0', 'ecc_errors/volatile/dram_uncorrectable': '0'}.items():
        assert g.findtext(key) == value, (key, g.findtext(key))
    return g
