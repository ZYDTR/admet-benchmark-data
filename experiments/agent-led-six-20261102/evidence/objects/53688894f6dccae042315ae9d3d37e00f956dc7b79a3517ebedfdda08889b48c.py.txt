import datetime, hashlib, json, os, stat, uuid
from pathlib import Path
ROOT = Path(os.environ['ADMET_RUN_ROOT'])
GPU_UUID = os.environ['CUDA_VISIBLE_DEVICES']

def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def safe(path, existing=False):
    path = Path(path)
    if ROOT.resolve(strict=True) != ROOT or not (path == ROOT or ROOT in path.parents) or path.resolve(strict=existing) != path:
        raise ValueError('noncanonical task path')
    for parent in (path, *path.parents):
        if parent.exists() or parent.is_symlink():
            s = parent.lstat()
            if stat.S_ISLNK(s.st_mode) or (stat.S_ISREG(s.st_mode) and s.st_nlink != 1):
                raise ValueError('linked task path')
    return path

def sha(path):
    h = hashlib.sha256()
    with safe(path, True).open('rb') as stream:
        for v in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(v)
    return h.hexdigest()

def write_json(path, value):
    path = safe(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + '.' + uuid.uuid4().hex + '.tmp')
    with safe(tmp).open('x') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write('\n')
        stream.flush()
        os.fsync(stream.fileno())
    tmp.replace(path)
