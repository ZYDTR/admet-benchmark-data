+"""Read-only cache provenance anchored outside the mutable cache itself.
+
+The expected manifest SHA must come from the controller's frozen task, never
+from hashing an untrusted manifest and accepting that same value as authority.
+"""
+from __future__ import annotations
+
+import json
+from pathlib import Path
+from types import MappingProxyType
+
+from .common import InterfaceError, sha256_file
+
+
+class CacheTrust:
+    def __init__(self, manifest_path, expected_sha256,
+                 prefix="input/legacy/cache/chemeleon_mp/"):
+        path = Path(manifest_path)
+        if sha256_file(path) != expected_sha256:
+            raise InterfaceError("trusted manifest digest mismatch")
+        manifest = json.loads(path.read_text(encoding="utf-8"))
+        entries = {}
+        for row in manifest["files"]:
+            name = row["path"]
+            if name.startswith(prefix):
+                relative = name[len(prefix):]
+                if relative in entries or ".." in Path(relative).parts:
+                    raise InterfaceError("invalid trusted cache manifest member")
+                entries[relative] = (row["sha256"], row["bytes"])
+        if not entries:
+            raise InterfaceError("trusted cache manifest has no cache entries")
+        self.entries = MappingProxyType(entries)
+        self.manifest_sha256 = expected_sha256
+
+    def verify(self, root, task, split):
+        for name in (f"raw_{split}.npy", f"mask_{split}.npy",
+                     f"raw_{split}.meta.json"):
+            key = f"{task}/{name}"
+            if key not in self.entries:
+                raise InterfaceError(f"trusted cache entry missing: {key}")
+            path = Path(root) / key
+            expected_sha, expected_bytes = self.entries[key]
+            if not path.is_file():
+                raise InterfaceError(f"trusted cache file missing: {key}")
+            if path.stat().st_size != expected_bytes or sha256_file(path) != expected_sha:
+                raise InterfaceError(f"trusted cache digest mismatch: {key}")
+        return self.manifest_sha256

+def checked_predictions(records, rows, task, split, parse_mask=None):
+    if len(records) != len(rows):
+        raise InterfaceError("diagnostic prediction row count mismatch")
+    values = []
+    for index, (pred, row) in enumerate(zip(records, rows)):
+        expected = (task, split, str(index), row["Drug_ID"], row["Drug"])
+        actual = tuple(str(pred.get(key, "")) for key in
+                       ("task", "split", "row_index", "Drug_ID", "Drug"))
+        if actual != expected:
+            raise InterfaceError(f"diagnostic row identity mismatch at {index}")
+        if parse_mask is not None:
+            parsed = str(pred.get("parse_ok", "")).lower()
+            if parsed not in ("true", "false") or (parsed == "true") != bool(parse_mask[index]):
+                raise InterfaceError(f"diagnostic parse mask mismatch at {index}")
+        value = float(pred["y_pred"])
+        if not math.isfinite(value):
+            raise InterfaceError("nonfinite diagnostic prediction")
+        values.append(value)
+    return values
+

+def raw_statistics(features):
+    import numpy as np
+    x = np.asarray(features, dtype=np.float32)
+    sd32 = x.std(axis=0, ddof=0)
+    sd64 = x.astype(np.float64).std(axis=0, ddof=0)
+    return {
+        "raw_std32_min": float(sd32.min()),
+        "raw_std64_min": float(sd64.min()),
+        "exact_constant_dimensions": int(np.all(x == x[0], axis=0).sum()),
+        "low_std32_dimensions": int((sd32 < 1e-8).sum()),
+        "low_std64_dimensions": int((sd64 < 1e-8).sum()),
+    }
+
+
+def pearson(a, b):
+    import numpy as np
+    a, b = np.asarray(a, np.float64), np.asarray(b, np.float64)
+    if len(a) != len(b) or len(a) < 2 or not (np.isfinite(a).all() and np.isfinite(b).all()):
+        raise InterfaceError("invalid association inputs")
+    if np.ptp(a) == 0 or np.ptp(b) == 0:
+        return None
+    return float(np.corrcoef(a, b)[0, 1])

+                                ("prediction_deviation_from_median", deviation)]}
